cmake_minimum_required(VERSION 3.20)
project(SatCompSC09Opaque LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OFX_INC "C:/ProgramData/Blackmagic Design/DaVinci Resolve/Support/Developer/OpenFX/OpenFX-1.4/include" CACHE PATH "Installed Blackmagic SDK")
add_library(SatCompSC09Opaque MODULE constant_ofx.cpp)
add_library(SatCompSC09Harness MODULE constant_ofx.cpp)
target_compile_definitions(SatCompSC09Harness PRIVATE SAT19_STANDALONE_TEST)
foreach(t SatCompSC09Opaque SatCompSC09Harness)
  target_compile_definitions(${t} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${t} PRIVATE /W4 /fp:strict /O2)
  target_include_directories(${t} PRIVATE "${OFX_INC}")
endforeach()
set(bundle_dir "${CMAKE_BINARY_DIR}/SatCompSC09Opaque.ofx.bundle/Contents/Win64")
set_target_properties(SatCompSC09Opaque PROPERTIES PREFIX "" SUFFIX ".ofx" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${bundle_dir}" LIBRARY_OUTPUT_DIRECTORY_RELEASE "${bundle_dir}")
